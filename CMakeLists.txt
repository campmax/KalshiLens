cmake_minimum_required(VERSION 3.16)
project(KalshiLens VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(kalshilens src/main.cpp)
add_executable(kalshilens_tests tests/tests.cpp)
add_executable(monitor_tests tests/monitor_tests.cpp)
if(WIN32)
  target_link_libraries(kalshilens PRIVATE winhttp shell32)
  target_link_libraries(kalshilens_tests PRIVATE winhttp)
  target_link_libraries(monitor_tests PRIVATE winhttp)
else()
  find_package(CURL REQUIRED)
  target_link_libraries(kalshilens PRIVATE CURL::libcurl)
  target_link_libraries(kalshilens_tests PRIVATE CURL::libcurl)
  target_link_libraries(monitor_tests PRIVATE CURL::libcurl)
endif()
foreach(target kalshilens kalshilens_tests monitor_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /bigobj)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME accounting_and_api COMMAND kalshilens_tests)

add_test(NAME robust_monitor COMMAND monitor_tests)
